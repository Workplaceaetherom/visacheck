   575→
   576→```js
   577→import rules from "../../data/rules.json";
   578→import jsonLogic from "json-logic-js";
   579→
   580→const json=(o,s=200)=>new Response(JSON.stringify(o),{status:s,headers:{"content-type":"application/json"}});
   581→const isMissing=v=>v===undefined||v===null||v==="";
   582→
   583→function extractVars(o,out=[]){
   584→  if(o&&typeof o==="object"){
   585→    if(o.var!==undefined)out.push(o.var);
   586→    else for(const k in o)extractVars(o[k],out);
   587→  }
   588→  return [...new Set(out)];
   589→}
   590→
   591→async function verifyTurnstile(env,token,ip){
   592→  if(!env.TURNSTILE_SECRET)return true;
   593→  const r=await fetch("https://challenges.cloudflare.com/turnstile/v0/siteverify",{
   594→    method:"POST",headers:{"content-type":"application/json"},
   595→    body:JSON.stringify({secret:env.TURNSTILE_SECRET,response:token,remoteip:ip})});
   596→  return (await r.json()).success===true;
   597→}
   598→
   599→async function rateLimit(env,ip){
   600→  if(!env.RATE)return true;
   601→  const key=`rl:${ip}:${Math.floor(Date.now()/3600000)}`;
   602→  const n=parseInt(await env.RATE.get(key)||"0",10)+1;
   603→  await env.RATE.put(key,String(n),{expirationTtl:3700});
   604→  return n<=10;
   605→}
   606→
   607→export async function onRequestPost({request,env}){
   608→  try{
   609→    const ip=request.headers.get("cf-connecting-ip")||"anon";
   610→    if(!await rateLimit(env,ip))return json({error:"rate_limited"},429);
   611→    const body=await request.json();
   612→    if(body.website)return json({ok:true});
   613→    if(!/^[A-Z]{2}$/.test(body.iso||""))return json({error:"bad_input"},400);
   614→    if(JSON.stringify(body.profile||"").length>4000)return json({error:"bad_input"},400);
   615→    if(!await verifyTurnstile(env,body.turnstileToken||"",ip))
   616→      return new Response("Blocked",{status:403});
   617→
   618→    const allowed=new Set();
   619→    rules.countries.forEach(c=>c.categories.forEach(cat=>cat.rules.forEach(r=>extractVars(r.logic_json).forEach(v=>allowed.add(v)))));
   620→    const profile={};
   621→    for(const[k,v]of Object.entries(body.profile||{})){
   622→      if(!allowed.has(k))continue;
   623→      profile[k]=(typeof v==="string"&&/^-?\d+(\.\d+)?$/.test(v))?Number(v):v;
   624→    }
   625→
   626→    const country=rules.countries.find(c=>c.iso===body.iso);
   627→    if(!country)return json({error:"not_covered"},404);
   628→
   629→    const now=Date.now();
   630→    const results=country.categories.map(cat=>{
   631→      const checked=cat.rules
   632→        .filter(r=>new Date(r.valid_from).getTime()<=now&&(!r.valid_to||new Date(r.valid_to).getTime()>=now))
   633→        .filter(r=>!JSON.stringify(r).includes("PENDING_HUMAN_CLICK"))
   634→        .map(r=>{
   635→          const vars=extractVars(r.logic_json);
   636→          const missing=vars.filter(v=>isMissing(profile[v]));
   637→          if(missing.length)return{rule_id:r.rule_id,var:vars[0],description:r.description,
   638→            status:"incomplete",ui_note:r.ui_note||null,source_url:r.source_url};
   639→          const passed=!!jsonLogic.apply(r.logic_json,profile);
   640→          return{rule_id:r.rule_id,var:vars.find(v=>MONEY_LIKE(v))||vars[0],description:r.description,
   641→            summary:r.summary||null,status:passed?"passed":"failed",
   642→            entered:passedValue(r,profile),ui_note:r.ui_note||null,source_url:r.source_url};
   643→        });
   644→      const allPassed=checked.length>0&&checked.every(c=>c.status==="passed");
   645→      const hasInc=checked.some(c=>c.status==="incomplete");
   646→      return{category_id:cat.category_id,visa_name:cat.visa_name,
   647→        official_guidance_url:cat.official_guidance_url,
   648→        overall_status:checked.length===0?"unavailable":allPassed?"passed":hasInc?"incomplete":"failed",
   649→        checks:checked};
   650→    });
   651→
   652→    return json({country_iso:country.iso,evaluated_at:new Date().toISOString(),results});
   653→  }catch(e){
   654→    console.error("check failed:",e);
   655→    return json({error:"internal"},500);
   656→  }
   657→}
   658→const MONEY_LIKE=v=>/gbp|balance|income|rate/i.test(v);
   659→function passedValue(r,profile){
   660→  const s=JSON.stringify(r.logic_json);
   661→  for(const v of extractVars(r.logic_json)){
   662→    if(MONEY_LIKE(v)&&profile[v]!==undefined)
   663→      return s.includes('"'+v+'"')&&typeof profile[v]==="number"?profile[v]:profile[v];
   664→  }
   665→  return undefined;
   666→}
