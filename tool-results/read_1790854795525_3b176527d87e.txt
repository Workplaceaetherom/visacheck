    10→
    11→export type RuleCategory = 'skilled_worker' | 'student' | 'visitor' | 'family' | 'business' | 'investor';
    12→
    13→export interface RuleAuthority {
    14→  shortName: string;          // e.g. "UKVI", "USCIS", "IRCC", "Home Affairs", "BAMF", "ICP"
    15→  officialName: string;       // e.g. "UK Visas and Immigration"
    16→  url: string;                // https URL to the official guidance
    17→}
    18→
    19→export interface ProposedChange {
    20→  status: 'consultation' | 'proposed' | 'announced' | 'deferred';
    21→  summary: string;            // human-readable description of the upcoming change
    22→  expectedEffectiveAt: string; // ISO date string
    23→  sourceUrl: string;           // link to the announcement
    24→}
    25→
    26→export interface Rule {
    27→  id: string;                  // unique within (countryIso, category)
    28→  countryIso: string;          // "GB" | "US" | "CA" | "AU" | "DE" | "AE"
    29→  category: RuleCategory;
    30→  // The status marker — gate is: JSON.stringify(rule).includes('PENDING_HUMAN_CLICK')
    31→  status: 'PENDING_HUMAN_CLICK';
    32→  authority: RuleAuthority;
    33→  lastUpdated: string;         // ISO date the rule data was last reviewed
    34→  effectiveFrom: string;       // ISO date the rule became effective
    35→  validTo?: string | null;     // ISO date the rule expires (null = open-ended)
    36→  title: { en: string; ur: string; bn: string };
    37→  summary: { en: string; ur: string; bn: string };
    38→  source: string;              // https URL to the official published rule
    39→  proposedChange?: ProposedChange;
    40→  // Pure logical evaluator. Result is masked by the gate — if status is PENDING_HUMAN_CLICK,
    41→  // the user always sees "pending" (the safe "being verified" state).
    42→  applies: (a: Answers) => boolean;
    43→  evaluate: (a: Answers) => 'pass' | 'fail' | 'na';
    44→}
    45→

    47→  nationality: string;            // ISO country name
    48→  applyingFromOutsideUK: boolean;
    49→  jobTitle: string;
    50→  socCode: string;
    51→  salary: number;                 // gross annual, in destination currency
    52→  hasCos: boolean;                // certificate of sponsorship / job offer
    53→  sponsorLicensed: 'yes' | 'unsure' | 'no';
    54→  newEntrant: boolean;
    55→  englishMet: boolean;
    56→  fundsMet: boolean;
    57→  ihsAware: boolean;
    58→  criminalCert: boolean;
    59→  passportValid: boolean;
    60→  // Extended for other visa categories
    61→  admissionOffer: boolean;        // student: has school admission
    62→  studyFundsMet: boolean;         // student: tuition + living funds
    63→  relationshipProof: boolean;     // family: proof of relationship to settled person
    64→  sponsorIncomeMet: boolean;      // family: sponsor meets income threshold
    65→  businessPlan: boolean;          // business/investor: has business plan
    66→  investmentAmount: number;       // investor: capital to invest (in destination currency)
    67→  visitPurposeValid: boolean;     // visitor: stated purpose allowed
    68→  returnTicket: boolean;          // visitor: has return/onward ticket
    69→  healthInsurance: boolean;        // visitor: has travel insurance
    70→}
